git init — создаёт новый локальный репозиторий в текущей папке
git clone <repo_url> — копирует существующий удалённый репозиторий на компьютер
git status — показывает текущее состояние файлов
git add <file> — добавляет конкретный файл в индекс
git add . — добавляет все изменённые файлы в индекс
git commit -m "message" — сохраняет изменения в историю с сообщением
git branch — показывает список веток
git branch <branch> — создаёт новую ветку
git checkout <branch> — переключается на указанную ветку
git merge <branch> — объединяет указанную ветку с текущей
git pull origin <branch> — скачивает последние изменения и объединяет их
git push origin <branch> — отправляет локальные коммиты на GitHub
git remote -v — показывает привязанные удалённые репозитории
git diff — показывает разницу между изменениями
git log — показывает полную историю коммитов
git log --oneline — показывает сокращённую историю коммитов
git revert <commit> — отменяет указанный коммит новым коммитом
git reset <commit> — сбрасывает состояние до указанного коммита
git tag — показывает список тегов
git tag -a v1.0 -m "tag" — создаёт тег с описанием
git config --global user.name "name" — устанавливает имя пользователя
git config --global user.email "email" — устанавливает email пользователя
